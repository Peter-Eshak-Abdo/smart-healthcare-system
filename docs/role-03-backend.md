```ts
export function buildApp(deps: AppDeps) {
  const app = express();
  app.disable('x-powered-by');
  app.set('trust proxy', 1);                                  // behind one proxy (Railway/Render)

  app.use(requestId());                                       // 1. X-Request-Id
  app.use(httpLogger(deps.logger));                           // 2. pino-http with redaction
  app.use(securityHeaders());                                 // 3. helmet + HSTS + no-store on /v1

  app.use('/v1', systemRoutes());                             //    health, ready, .well-known, config (public)
  app.use('/v1/auth', rateLimit('auth_ip'));                  // 6. IP limiter on auth endpoints

  app.use('/v1', authenticate(deps));                         // 7. JWT → AuthContext (status, epoch, licence)
  app.use('/v1', rateLimit('global_user'));                   // 8. per-user global limiter
  // 9. per-route: requirePermission → (requireStepUp) → (rateLimit bucket) → idempotency → validate → controller

  app.use(errorHandler(deps.logger));                         // 11. AppError/ZodError/PG → RFC 9457
```

### 4.1 `authenticate`

```ts
export const authenticate = (deps: AppDeps): RequestHandler => async (req, res, next) => {
  const raw = req.header('authorization');
  if (!raw?.startsWith('Bearer ')) throw new AppError('AUTH_TOKEN_MISSING');
  const claims = await verifyAccessToken(raw.slice(7)).catch(mapJoseError);   // EXPIRED / INVALID
  const ctx = await deps.authContext.load(claims.sub);                          // Redis 30 s → DB
  if (!ctx) throw new AppError('AUTH_TOKEN_INVALID');
  if (ctx.status !== 'active') throw new AppError('ACCOUNT_SUSPENDED');
  if (claims.session_epoch < ctx.sessionEpoch || claims.app_role !== ctx.role)
    throw new AppError('AUTH_SESSION_REVOKED');
  if (STAFF_ROLES.has(ctx.role) && claims.aal !== 'aal2') throw new AppError('AUTH_MFA_REQUIRED');
  if (ctx.role !== 'admin' && ctx.role !== 'patient' && !ctx.licenceValid) throw new AppError('LICENSE_NOT_VALID');
  res.locals.auth = { claims, ...ctx, permissions: new Set(ROLE_PERMISSIONS[ctx.role]) };
  next();
};
```

Note: a patient with `status = pending_verification` may call only `GET /v1/me`, `POST /v1/patients/me/onboarding`, and `POST /v1/patients/me/consents`. These routes use `authenticate({ allowPending: true })`.

### 4.2 `errorHandler`

```ts
export const errorHandler = (logger: Logger): ErrorRequestHandler => (err, req, res, _next) => {
  const appErr = toAppError(err);          // AppError | ZodError→VALIDATION_FAILED | PgError→role-02 §8 | unknown→INTERNAL_ERROR
  if (appErr.status >= 500) logger.error({ err: redact(err), rid: res.locals.requestId }, 'unhandled');
  if (appErr.audit) void auditDenied(req, res, appErr);   // ACCESS_DENIED for 403/404 on PHI routes
  res.status(appErr.status).type('application/problem+json').json({
    type: `https://docs.smarthealth.app/errors/${appErr.code}`,
    title: t(req, `errors.${appErr.code}.title`),
    status: appErr.status, code: appErr.code,
    detail: appErr.detail, instance: req.originalUrl.split('?')[0],
    request_id: res.locals.requestId, ...(appErr.errors && { errors: appErr.errors }),
    ...(appErr.extra ?? {}),
  });
};
```

## 5. Database Access: `withUserTx`

```ts
// src/db/with-user-tx.ts
export async function withUserTx<T>(
  auth: AuthLocals,
  fn: (tx: Transaction<DB>) => Promise<T>,
  opts: { pool?: 'default' | 'emergency'; isolation?: 'read committed' | 'repeatable read' } = {},
): Promise<T> {
  const db = opts.pool === 'emergency' ? emergencyDb : defaultDb;
  return db.transaction().setIsolationLevel(opts.isolation ?? 'read committed').execute(async (tx) => {
    await sql`set local role authenticated`.execute(tx);
    await sql`select set_config('request.jwt.claims', ${JSON.stringify(auth.claims)}, true)`.execute(tx);
    await sql`set local statement_timeout = '8s'`.execute(tx);
    return fn(tx);
  });
}
```

Rules (lint-enforced, see ANTIGRAVITY-RULES):
- Domain code may only access the DB through `withUserTx`.
- The only exception is `src/workers/*` and `scripts/*`, which run as `postgres` for jobs and maintenance.
- `supabase-admin.ts` (service key) may be imported only from `modules/admin/*`, `modules/auth/*` (signOut), and `scripts/seed-dev.ts`.
- Retries: on `40P01`/`40001`, retry the whole `withUserTx` callback once.

```ts
async onboard(auth: AuthLocals, input: OnboardingRequest) {
  validateEgyptianNationalId(input.id_document_number, input.date_of_birth);       // E-ID-02
  const bidx = blindIndex(`${input.id_document_type}:${input.id_document_number}`, 'id_document');
  const enc  = encryptField(input.id_document_number, { table: 'patients', column: 'id_document_enc', rowId: auth.claims.sub });

        ${input.blood_type ?? null}::app.blood_type, ${input.consents[0].version})`.execute(tx);

```

`validateEgyptianNationalId`: 14 digits. Digit 1 is the century (2 → 1900s, 3 → 2000s). Digits 2–7 are YYMMDD and must match the DOB. Digits 8–9 are a valid governorate code (01–04, 11–19, 21–35, 88).

### 6.2 Share codes & grants
- `POST /patients/me/share-codes`: generate a 6-digit code with `crypto.randomInt(0, 1_000_000)`, zero-pad it, store `HMAC(BLIND_INDEX_KEY, "share:" + patientId + ":" + code)`, and return the code once.
- `POST /doctor/access/redeem`: compute the same HMAC, then `select private.redeem_share_code(mrn, hmac)`. `NULL` → `422 invalid_or_expired_code` and consume a `share_code_fail` point. On success: audit `ACCESS_GRANTED`.

### 6.3 Prescription issue (Backend B)

```ts
// src/modules/prescriptions/service.ts
async issue(auth: AuthLocals, req: IssuePrescriptionRequest, idemKey: string) {
  if (nowSec() - auth.claims.auth_time > 15 * 60) throw new AppError('AUTH_STEP_UP_REQUIRED', { detail: 'reauth_totp' });

  return withUserTx(auth, async (tx) => {
    // 1) Load encounter (RLS: must be this doctor's, open or closed <24h) and patient context
    const enc = await repo.getEncounterForPrescribing(tx, req.encounter_id, req.patient_id);
    if (!enc) throw new AppError('ACCESS_GRANT_REQUIRED');
    const ctx = await repo.loadSafetyContext(tx, req.patient_id);   // active meds, allergies, conditions (all visibilities)
    const drugs = await repo.loadDrugs(tx, req.items.map(i => i.drug_id));
    validateItems(req.items, drugs);                                 // active drugs, units = base_unit, controlled rules

    // 2) Authoritative safety check (E-CLIN-03)
    const check = await safety.check(ctx, req.items);               // calls ai-service; handles unavailable
    const blocking = check.alerts.filter(a => a.severity === 'major' || a.severity === 'contraindicated');
    const missing = blocking.filter(a => !req.overrides.some(o => o.alert_key === a.alert_key && isValidJustification(o)));
    if (check.checks_status === 'unavailable' && !req.manual_review_attested)
      throw new AppError('RX_ALERT_OVERRIDE_REQUIRED', { detail: 'manual_review_attestation_required' });
    if (missing.length) throw new AppError('RX_ALERT_OVERRIDE_REQUIRED', {
      errors: missing.map(a => ({ path: 'overrides', code: 'missing', message: a.alert_key, alert_key: a.alert_key })) });

    // 3) Insert header + items (ids generated here so AAD/signature can include them)
    const rxId = randomUUID();
    const isControlled = drugs.some(d => d.is_controlled);
    const paperCode = generatePaperCode();                           // 6 chars Crockford, shown once to patient
    const header = await repo.insertPrescription(tx, {
      id: rxId, patient_id: req.patient_id, doctor_id: auth.claims.sub, facility_id: enc.facility_id,
      encounter_id: enc.id, expires_at: addDays(now(), isControlled ? 7 : 30),
      notes_to_pharmacist_enc: req.notes_to_pharmacist ? encryptField(req.notes_to_pharmacist, aad('prescriptions','notes_to_pharmacist_enc',rxId)) : null,
      safety_check_bypassed: check.checks_status === 'unavailable',
      verification_code_hash: await argon2.hash(paperCode, ARGON2_PARAMS),
    });
    const items = await repo.insertItems(tx, rxId, req.items.map(i => ({ id: randomUUID(), ...i,
      substitution_allowed: isControlledDrug(i, drugs) ? false : i.substitution_allowed })));
    await repo.insertAlerts(tx, rxId, req.patient_id, auth.claims.sub, check, req.overrides);

    // 4) Sign canonical payload (role-01 §6.4, 03 §7.5) and store — must happen before COMMIT
    const payload = buildSigningPayload(header, items, auth.practitioner.license_number);
    const { signature, kid, sha256 } = signer.signRx(payload);
    await repo.setSignature(tx, rxId, signature, kid, sha256);

    await audit(tx, 'RX_ISSUED', { resource: ['prescription', rxId], patient: req.patient_id,
                                    meta: { rx_number: header.rx_number, item_count: items.length } });
    for (const o of req.overrides) await audit(tx, 'ALERT_OVERRIDDEN', { resource: ['prescription', rxId],
      patient: req.patient_id, meta: { alert_key: o.alert_key, reason_code: o.reason_code } });
    return { detail: await repo.getDetail(tx, rxId), paperCode };
  });
}
```

`isValidJustification`: length ≥ 15, ≥ 3 words, not one repeated character (E-CLIN-02).

### 6.4 Pharmacy verify (Backend B)

```ts
async verify(auth: AuthLocals, req: VerifyRequest) {
  let rxId: string; let method: VerificationMethod; let jti: string | null = null;

  if (req.mode === 'pickup_token') {
    const tok = await signer.verifyPickupToken(req.pickup_token);         // EdDSA, exp, typ=pickup
    if (!tok) throw new AppError('PICKUP_TOKEN_INVALID');
    if (await repo.isPickupJtiConsumed(tok.jti)) throw new AppError('PICKUP_TOKEN_INVALID');
    rxId = tok.rx; method = 'pickup_token'; jti = tok.jti;
  } else {
    const found = await withUserTx(auth, tx => sqlFindRxForPharmacy(tx, req.rx_number));
    if (!found || !(await argon2.verify(found.verification_code_hash, normaliseCode(req.verification_code))))
      { await rateLimitFail('rx_verify_fail', req.pharmacy_id); throw new AppError('RX_NOT_FOUND'); }
    await presence.assert(req.presence, found);                           // sms_otp | face_1to1 → else 202/401
    rxId = found.prescription_id; method = req.presence.method;
  }

  return withUserTx(auth, async (tx) => {
    const v = await sql`select * from private.open_pharmacy_context(${rxId}, ${req.pharmacy_id},
                        ${method}::app.verification_method, ${jti})`.execute(tx).then(r => r.rows[0]);
    const detail = await repo.getDetailForPharmacy(tx, rxId);            // now visible via RLS
    const sigOk = signer.verifyRxSignature(buildSigningPayloadFromRows(detail), detail.signature, detail.signature_kid);
    if (!sigOk) {                                                        // E-RX-06 insider tampering
      await audit(tx, 'RX_VERIFY_FAILED', { resource: ['prescription', rxId], outcome: 'denied', reason: 'payload_mismatch' });
      await alerts.critical('rx_payload_mismatch', { rxId });
      throw new AppError('RX_SIGNATURE_INVALID', { detail: 'payload_mismatch' });
    }
    await audit(tx, 'RX_VERIFIED', { resource: ['prescription', rxId], patient: detail.patient_id, meta: { result: v.result, method } });
    return toVerifyResponse(v, detail);
  });
}
```

### 6.5 Dispense engine: the transactional core (Backend B)

Guarantees (proved by DB test T16): no over-dispensing under concurrency, idempotent retries, and status derived by triggers.

```ts
// src/modules/pharmacy/dispense.service.ts
async dispense(auth: AuthLocals, rxId: string, req: DispenseRequest, idemKey: string) {
  return withRetryOnce(() => withUserTx(auth, async (tx) => {
    // 0) Idempotency at DB level (Redis layer already handled fast replays)
    const prior = await repo.findDispensationByIdemKey(tx, idemKey);
    if (prior) return replay(prior);

    // 1) Lock header first, then items in id order → deterministic lock order, no deadlocks
    const rx = await sql<RxLockRow>`select id, status, expires_at, prescriber_invalid, version
                                      from app.prescriptions where id = ${rxId} for update`.execute(tx).then(r => r.rows[0]);
    if (!rx) throw new AppError('RESOURCE_NOT_FOUND');                     // no context → RLS hides → 404/403
    const items = await sql<ItemLockRow>`select id, drug_id, quantity_prescribed, quantity_dispensed, status, dispense_count
                                           from app.prescription_items where prescription_id = ${rxId}
                                          order by id for update`.execute(tx).then(r => r.rows);

    // 2) Business validation under lock (clean error codes; DB CHECK remains the last line)
    const v = await repo.getValidVerification(tx, req.verification_id, rxId);
    if (!v) throw new AppError('RBAC_FORBIDDEN', { detail: 'verification_context_expired' });
    if (rx.prescriber_invalid) throw new AppError('RX_PRESCRIBER_INVALID');
    if (rx.status === 'cancelled') throw new AppError('RX_CANCELLED');
    if (rx.status === 'expired' || rx.expires_at <= new Date()) throw new AppError('RX_EXPIRED');
    if (rx.status === 'dispensed') throw new AppError('RX_ALREADY_DISPENSED');

    const errors: FieldError[] = [];
    for (const [i, line] of req.items.entries()) {
      const it = items.find(x => x.id === line.prescription_item_id);
      if (!it) { errors.push({ path: `items[${i}].prescription_item_id`, code: 'not_found' }); continue; }
      const remaining = Number(it.quantity_prescribed) - Number(it.quantity_dispensed);
      if (it.status === 'cancelled') errors.push({ path: `items[${i}]`, code: 'cancelled' });
      else if (remaining <= 0) errors.push({ path: `items[${i}]`, code: 'already_dispensed' });
      else if (line.quantity > remaining) errors.push({ path: `items[${i}].quantity`, code: 'too_big', message: `Max ${remaining}` });
      if (line.substituted_drug_id) await assertEquivalentSubstitute(tx, it.drug_id, line.substituted_drug_id, errors, i);
      await assertControlledRules(tx, it, v.pharmacy_id, errors, i);      // E-RX-13
    }
    if (errors.length) throw new AppError(
      errors.every(e => e.code === 'already_dispensed') ? 'RX_ALREADY_DISPENSED' : 'RX_QUANTITY_EXCEEDED', { errors });

    // 3) Write. Triggers increment quantities and recompute statuses atomically.
    const dispId = randomUUID();
    await repo.insertDispensation(tx, { id: dispId, prescription_id: rxId, pharmacy_id: v.pharmacy_id,
      pharmacist_id: auth.claims.sub, verification_id: v.id, verification_method: v.method,
      idempotency_key: idemKey, pickup_jti: v.pickup_jti, counselling_given: req.counselling_given,
      notes_enc: req.notes ? encryptField(req.notes, aad('dispensations','notes_enc',dispId)) : null });
    await repo.insertDispensationItems(tx, dispId, req.items);
    await sql`delete from app.dispense_leases where prescription_id = ${rxId}`.execute(tx);

    const after = await repo.getDispenseResult(tx, rxId, dispId);
    await audit(tx, 'RX_DISPENSED', { resource: ['prescription', rxId], patient: after.patient_id,
                                      meta: { dispensation_id: dispId, items: req.items.length } });
    await outbox(tx, after.patient_id, 'prescription_dispensed', { rx_number: after.rx_number });
    return after;
  }));
}
```

The `pickup_jti` UNIQUE constraint makes a pickup token single-use even across pharmacies. The `idempotency_key` UNIQUE constraint makes retries safe even when Redis is down.

### 6.6 Emergency identify & scoping (Backend B)

```ts
async identify(auth: AuthLocals, frames: Buffer[], form: IdentifyForm, idemKey: string) {
  const analysis = await ai.faceAnalyze({ mode: 'identify', liveness_mode: 'passive', frames });   // throws AI_* on failure
  await this.recordAttemptAndMapFailures(auth, analysis);          // BIO_NO_FACE/MULTIPLE/LOW_QUALITY/LIVENESS_FAILED

  return withUserTx(auth, async (tx) => {
    const top = await sql<{ patient_id: string; similarity: number }>`
      select * from private.match_face(${toPgVector(analysis.template.vector)}::extensions.vector,
                                       ${analysis.model_version}, ${analysis.transform_key_version}, 5)`
      .execute(tx).then(r => r.rows);
    const [first, second] = top;
    const margin = first ? first.similarity - (second?.similarity ?? 0) : 0;
    const decision = decideMatch(first?.similarity, margin);       // 'match' | 'possible_match' | 'no_match'
    await sql`select private.record_biometric_attempt('identify', ${decision}, ${first?.similarity ?? null},
              ${second?.similarity ?? null}, ${analysis.liveness.score}, ${analysis.model_version})`.execute(tx);
    await audit(tx, 'FACE_IDENTIFY_ATTEMPT', { meta: { result: decision, liveness_passed: true } });

    if (decision === 'no_match') return { result: 'no_match', fallbacks: ['emergency_card_qr', 'national_id_search'] };
    if (decision === 'possible_match') return this.issueCandidateTokens(tx, top.slice(0, 2));   // sex + age band only; Redis 120 s
    const session = await openSession(tx, first!.patient_id, 'face', first!.similarity, margin, false, false, form, idemKey);
    return { result: 'match', session: pick(session, ['id','expires_at','low_confidence']), patient_preview: await preview(tx, session) };
  }, { pool: 'emergency' });
}

export function decideMatch(sim: number | undefined, margin: number) {
  if (sim === undefined || sim < env.FACE_MATCH_POSSIBLE) return 'no_match';
  if (sim >= env.FACE_MATCH_ACCEPT && margin >= env.FACE_MATCH_MARGIN) return 'match';
  return 'possible_match';
}
```

Profile/contacts endpoints call **only** `private.get_emergency_profile(session)` / `private.get_emergency_contacts(session)`. The response is parsed with a **strict** Zod schema (`EmergencyProfileResponse.strict()`), so an accidental extra field fails the request instead of leaking data. Contacts are decrypted in the backend before responding. Every read is audited.

### 6.7 Assistant proxy (Backend B)

1. Check `patient`, adult (`age ≥ 16`), consent `ai_assistant`, rate limit `assistant_msg`.
2. Load the last 10 messages of the session (decrypt), call `ai.triage({history, message, locale, patient_context: {age_band, sex}})` with a 9 s timeout.
3. On AI failure, call `ai.redFlagsOnly(message)`. The rules engine is mirrored in the backend as a lightweight fallback (`modules/assistant/red-flags.ts`, same YAML file as ai-service, loaded from `packages/contracts/red-flags.yaml`). If nothing fires, return the fallback text (US-13-AC4).
4. Persist the user and assistant messages encrypted (`content_enc`) with `triage_level`, `red_flag`, `rule_ids`, `specialty_code`.
5. If `red_flag`, audit `ASSISTANT_RED_FLAG` (rule IDs only).

## 8. Idempotency Middleware

```ts
export const idempotency = ({ required }: { required: boolean }): RequestHandler => async (req, res, next) => {
  const key = req.header('idempotency-key');
  if (!key) return required ? next(new AppError('IDEMPOTENCY_KEY_REQUIRED')) : next();
  if (!isUuid(key)) return next(new AppError('VALIDATION_FAILED', { detail: 'idempotency_key_format' }));
  const rkey = `idem:${res.locals.auth.claims.sub}:${req.method}:${req.route.path}:${key}`;
  const bodyHash = sha256(canonicalize(req.body ?? {}) + fileDigests(req));
  const existing = await redis.get(rkey);
  if (existing) {
    const rec = JSON.parse(existing) as IdemRecord;
    if (rec.bodyHash !== bodyHash) return next(new AppError('IDEMPOTENCY_KEY_CONFLICT'));
    if (rec.state === 'in_progress') return next(new AppError('IDEMPOTENCY_KEY_CONFLICT', { detail: 'in_progress' }));
    return res.status(rec.status).set('Idempotency-Replayed', 'true').json(rec.body);
  }
  await redis.set(rkey, JSON.stringify({ state: 'in_progress', bodyHash }), 'EX', 60, 'NX');
  res.locals.idempotencyKey = key;
  hookResponse(res, async (status, body) => {
    if (status < 500) await redis.set(rkey, JSON.stringify({ state: 'done', bodyHash, status, body }), 'EX', 86_400);
    else await redis.del(rkey);
  });
  next();
};
```

## 9. Notifications Outbox

- Producers insert into `app.notifications` **inside the business transaction**, so notifications are exactly as durable as the business change.
- `workers/outbox.ts` polls every 5 s as `postgres`: `select … where status='pending' order by created_at limit 50 for update skip locked`, sends via `notifier`, then marks `sent`/`failed` (max 5 attempts, exponential backoff).
